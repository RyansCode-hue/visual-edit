import { StrictMode } from 'react'
import { createRoot } from 'react-dom/client'
import { MantineProvider, createTheme, rem } from '@mantine/core'
import type { MantineColorsTuple } from '@mantine/core'
import { ModalsProvider } from '@mantine/modals'
import { Notifications } from '@mantine/notifications'
import '@mantine/core/styles.css'
import './index.css'
import App from './app.tsx'
import ErrorBoundary from './components/error-boundary.tsx'

// Atticus Backoffice design tokens (frozen spec, "Design tokens" section).
// One createTheme restyles every screen from here; per-page styling should
// never need to repeat a colour or a font.
//
// This is an internal research tool -- a clinical workbench a researcher
// lives in for hours, not a marketing surface: dense, calm, data-forward,
// crisp borders, restrained colour used only for status and action, no
// decorative gradients. Barlow carries body/UI copy; headings use Barlow
// Semi Condensed at 600 weight -- the spec's own fallback, since Effra is
// not a free web font and this keeps a distinct-but-compatible heading
// voice with no licensing dependency. Atticus brand blue (#0071ce) is
// primary; the secondary slate (#1c3144) is reserved for secondary text
// and nav chrome, never a competing brand colour. Compact density and a
// small radius keep dense donor tables/forms legible without feeling
// cramped. Colour names below keep the "atticus*" prefix the rest of the
// app already references; only the ramps/usage changed to match tokens.

// primary_hex #0071ce, index 6. Atticus brand blue.
const atticusInk: MantineColorsTuple = [
  '#E6F3FC',
  '#C2E1F8',
  '#99CCF2',
  '#6DB5EC',
  '#479FE6',
  '#2690E2',
  '#0071CE',
  '#005CB0',
  '#004A8F',
  '#00386B',
];

// secondary_hex #0b2f55. Deep navy -- secondary text and nav
// chrome, plus the "in progress" / type-label badge accent. Never a
// competing brand colour.
const atticusBrass: MantineColorsTuple = [
  '#E8EEF5',
  '#C9D6E5',
  '#9FB5D0',
  '#6D90B8',
  '#3F6C9E',
  '#1E4C80',
  '#0B2F55',
  '#092746',
  '#071F38',
  '#05162A',
];

// neutral_hex #f4f6f8. A crisp, quiet off-white surface for a records
// workbench used all day -- never a cold pure-white admin template, never
// a warm parchment either (this is clinical/precise, not archival).
const atticusParchment: MantineColorsTuple = [
  '#FFFFFF',
  '#FBFCFD',
  '#F4F6F8',
  '#E3E7EB',
  '#CBD2D9',
  '#AEB7C0',
  '#8D97A1',
  '#6B7580',
  '#4A535C',
  '#2B3238',
];

const theme = createTheme({
  colors: {
    atticusInk,
    atticusBrass,
    atticusParchment,
    // Kept as an alias so any page still reading the earlier "neutral"
    // token name resolves to the same ramp.
    atticusNeutral: atticusParchment,
  },
  primaryColor: 'atticusInk',
  primaryShade: 6,
  fontFamily: 'Barlow, "Helvetica Neue", Arial, sans-serif',
  fontFamilyMonospace:
    'ui-monospace, SFMono-Regular, Menlo, monospace',
  headings: {
    fontFamily: '"Barlow Semi Condensed", Barlow, sans-serif',
    fontWeight: '600',
  },
  defaultRadius: 'sm',
  // density: compact -- an internal backoffice tool used all day by
  // researchers, not a consumer product; crisp edges and tight spacing
  // reduce scroll and support fast scanning.
  spacing: {
    xs: rem(6),
    sm: rem(9),
    md: rem(13),
    lg: rem(18),
    xl: rem(26),
  },
  components: {
    Table: {
      defaultProps: {
        verticalSpacing: 'xs',
        fontSize: 'sm',
      },
    },
    Badge: {
      defaultProps: {
        radius: 'sm',
      },
    },
  },
})

createRoot(document.getElementById('root')!).render(
  <StrictMode>
    <MantineProvider theme={theme} defaultColorScheme="light" forceColorScheme="light">
      <ModalsProvider>
        <Notifications position="bottom-right" />
        <ErrorBoundary>
          <App />
        </ErrorBoundary>
      </ModalsProvider>
    </MantineProvider>
  </StrictMode>,
)
